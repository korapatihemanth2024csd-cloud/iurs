    // DOM Element selections
    const sectionRadios = document.querySelectorAll('input[name="section"]');
    const sectionDisplay = document.getElementById("sectionDisplay");
    const checkboxes = document.querySelectorAll(".student-checkbox");
    const submitBtn = document.getElementById("submitBtn");
    const presentCount = document.getElementById("presentCount");
    const absentCount = document.getElementById("absentCount");
    const attendancePercent = document.getElementById("attendancePercent");
    // Update section display immediately when radio selection changes using a while loop
    sectionRadios.forEach((radio) => {
        radio.addEventListener("change", () => {
            let i = 0;
            while (i < sectionRadios.length) {
                if (sectionRadios[i].checked) {
                    sectionDisplay.textContent = sectionRadios[i].value;
                    break;
                }
                i++;
            }
        });
    });
    // Calculate attendance metrics on button click
    submitBtn.addEventListener("click", () => {
        let present = 0;
        const total = checkboxes.length;
        // Count present students using a for loop
        for (let i = 0; i < checkboxes.length; i++) {
            if (checkboxes[i].checked) {
                present++;
            }
        }
        // Calculate absent count using a do...while loop
        let absent = 0;
        let count = present;
        if (total > present) {
            do {
                absent++;
                count++;
            } while (count < total);
        }
        // Calculate percentage
        const percentage = total > 0 ? ((present / total) * 100).toFixed(2) + "%" : "0%";
        // Update DOM elements
        presentCount.textContent = present;
        absentCount.textContent = absent;
        attendancePercent.textContent = percentage;
    });
});

            subjectDisplay.textContent = this.value;
        });

        let answers = ["a", "b", "c"];
        let score = 0;
        for (let i = 0; i < questions.length; i++) {
            for (let j = 0; j < questions[i].length; j++) {
                if (questions[i][j].checked && questions[i][j].value == answers[i]) {
                    score++;
                }
            }
        }
        quizScore.textContent = score;
        let grade = "-";
        let remark = "-";
        let count = 0;
        if (score == 3) {
            do {
                grade = "A";
                remark = "Excellent";
                count++;
            } while (count < 1);
        } else if (score == 2) {
            do {
                grade = "B";
                remark = "Good";
                count++;
            } while (count < 1);
        } else if (score == 1) {
            do {
                grade = "C";
                remark = "Needs Improvement";
                count++;
            } while (count < 1);
        } else {
            do {
                grade = "F";
                remark = "Please Try Again";
                count++;
            } while (count < 1);
        }

        quizRemark.textContent = remark;
        let selectedTopics = [];
        let k = 0;

        if (topics.length > 0) {
            do {
                if (topics[k].checked) {
                    selectedTopics.push(topics[k].value);
                }
                k++;
            } while (k < topics.length);
        }
        if (selectedTopics.length > 0) {
            topicList.textContent = selectedTopics.join(", ");
        } else {
            topicList.textContent = "-";
        }
    });

        for (let i = 0; i < q2radios.length; i++) {
            q2radios[i].checked = false;
        }
        for (let i = 0; i < q3radios.length; i++) {
            q3radios[i].checked = false;
        }
        for (let i = 0; i < topics.length; i++) {
