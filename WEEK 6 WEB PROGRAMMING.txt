      if (editor.style.fontWeight === 'bold') {

      }
    });

    });

      editor.style.fontWeight = 'normal';
      editor.style.fontStyle = 'normal';
  
      fontSizeSelect.value = '12px';
      fontColorSelect.value = 'black';

      italicBtn.classList.remove('active');

Script js: document.addEventListener('DOMContentLoaded', () => {
  const userForm = document.getElementById('userForm');
  const nameInput = document.getElementById('name');
  const emailInput = document.getElementById('email');
  const ageInput = document.getElementById('age');

  const nameError = document.getElementById('nameError');
  const emailError = document.getElementById('emailError');
  const ageError = document.getElementById('ageError');

  const successMessage = document.getElementById('successMessage');
  const successData = document.getElementById('successData');

  userForm.addEventListener('submit', (event) => {
    // Prevent default form re-submission / reload
    event.preventDefault();

    // Clear previous error/success states
    nameInput.classList.remove('is-invalid');
    emailInput.classList.remove('is-invalid');
    ageInput.classList.remove('is-invalid');
    successMessage.classList.add('d-none');

    const nameValue = nameInput.value.trim();
    const emailValue = emailInput.value.trim();
    const ageValue = parseInt(ageInput.value, 10);

    let isValid = true;

    // 1. Validate Name
    if (nameValue === '') {
      nameInput.classList.add('is-invalid');
      isValid = false;
    }
    // 2. Validate Email (Must have a valid domain ending in .com)
    const emailRegex = /^[a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+\.com$/;
    if (!emailRegex.test(emailValue)) {
     emailInput.classList.add('is-invalid');
    isValid = false;
    }

    // 3. Validate Age (Between 18 and 120)
    if (isNaN(ageValue) || ageValue < 18 || ageValue > 120) {
      ageInput.classList.add('is-invalid');
      isValid = false;
    }

    // If all inputs pass validation
    if (isValid) {
      // Show data in success container
      successData.textContent = `Name: ${nameValue}, Email: ${emailValue}, Age: ${ageValue}`;
      successMessage.classList.remove('d-none');

      // Reset form input fields
      userForm.reset();
    }
  });
});

Script js:   const canvas = document.getElementById("sketchPad");
const ctx = canvas.getContext("2d");

const redBtn = document.getElementById("redBtn");
const blueBtn = document.getElementById("blueBtn");
const greenBtn = document.getElementById("greenBtn");
const eraserBtn = document.getElementById("eraserBtn");
const clearBtn = document.getElementById("clearBtn");
const currentColor = document.getElementById("currentColor");
const currentColorDisplay = document.getElementById("currentColorDisplay");

function resetCanvasBackground() {
    ctx.fillStyle = "#ffffff";
    ctx.fillRect(0, 0, canvas.width, canvas.height);
}

resetCanvasBackground();

ctx.strokeStyle = "#000000";
ctx.lineWidth = 5;
ctx.lineCap = "round";

let isDrawing = false;

function setDrawingColor(displayText, hexColor) {
    ctx.strokeStyle = hexColor;
    if (currentColor) {
        currentColor.textContent = displayText;
    }
    if (currentColorDisplay) {
        currentColorDisplay.textContent = displayText;
    }
}

function getCanvasCoordinates(e) {
    const rect = canvas.getBoundingClientRect();
    return {
        x: e.clientX - rect.left,
        y: e.clientY - rect.top
    };
}

canvas.addEventListener("mousedown", (e) => {
    isDrawing = true;
    const coords = getCanvasCoordinates(e);
    ctx.beginPath();
    ctx.moveTo(coords.x, coords.y);
});

canvas.addEventListener("mousemove", (e) => {
    if (!isDrawing) return;
    const coords = getCanvasCoordinates(e);
    ctx.lineTo(coords.x, coords.y);
    ctx.stroke();
});

canvas.addEventListener("mouseup", () => {
    isDrawing = false;
    ctx.beginPath();
});

canvas.addEventListener("mouseout", () => {
    isDrawing = false;
    ctx.beginPath();
});

redBtn.addEventListener("click", () => setDrawingColor("Red", "#ff0000"));
blueBtn.addEventListener("click", () => setDrawingColor("Blue", "#0000ff"));
greenBtn.addEventListener("click", () => setDrawingColor("Green", "#008000"));
eraserBtn.addEventListener("click", () => setDrawingColor("White", "#ffffff"));

clearBtn.addEventListener("click", () => {
    resetCanvasBackground();
});
