1. src/components/PriceTag.jsx
import React, { useState } from "react";
import PropTypes from "prop-types";
import "./PriceTag.css";

function PriceTag({ productName, originalPrice }) {
  const [isDiscounted, setIsDiscounted] = useState(false);

  const displayedPrice = isDiscounted
    ? originalPrice * 0.9
    : originalPrice;

  const toggleDiscount = () => {
    setIsDiscounted(!isDiscounted);
  };

  return (
    <div id="price-tag">
      <h2 id="product-name">{productName}</h2>

      <p id="displayed-price">
        Price: ₹{displayedPrice}
      </p>

      <button id="toggle-discount" onClick={toggleDiscount}>
        {isDiscounted ? "Remove Discount" : "Apply Discount"}
      </button>
    </div>
  );
}

PriceTag.propTypes = {
  productName: PropTypes.string.isRequired,
  originalPrice: PropTypes.number.isRequired,
};

export default PriceTag;

import React, { useState, useEffect } from "react";
import PropTypes from "prop-types";
import "./Marks.css";

function Marks({ marks }) {
  const [totalScore, setTotalScore] = useState(0);

  useEffect(() => {
    const total = marks.reduce((sum, mark) => sum + mark.score, 0);
    setTotalScore(total);

    console.log("Visited Marks Page");
  }, [marks]);

  return (
    <div id="marks-container">
      <h1 id="marks-title">Subject Marks</h1>

      <div id="marks-list">
        {marks.map((mark, index) => (
          <div className="mark-item" key={index}>
            <span className="mark-subject">{mark.subject}</span>
            <span className="mark-score">Score: {mark.score}</span>
          </div>
        ))}
      </div>

      <p id="total-score">Total Score: {totalScore}</p>
    </div>
  );
}

Marks.propTypes = {
  marks: PropTypes.arrayOf(
    PropTypes.shape({
      subject: PropTypes.string,
      score: PropTypes.number,
    })
  ).isRequired,
};

export default Marks;

import React, { useState } from "react";
import EmployeeCard from "./components/EmployeeCard";
import "./App.css";

function App() {
  const [selectedDepartment, setSelectedDepartment] = useState("All");

  const employees = [
    {
      id: 1,
      employeeName: "Rajan Kumar",
      role: "Software Engineer",
      department: "Engineering",
    },
    {
      id: 2,
      employeeName: "Meena Iyer",
      role: "Product Manager",
      department: "Product",
    },
    {
      id: 3,
      employeeName: "Karthik Raj",
      role: "UI Designer",
      department: "Design",
    },
    {
      id: 4,
      employeeName: "Sneha Pillai",
      role: "QA Engineer",
      department: "Engineering",
    },
    {
      id: 5,
      employeeName: "Divya Nair",
      role: "Data Analyst",
      department: "Product",
    },
  ];

  const filteredEmployees =
    selectedDepartment === "All"
      ? employees
      : employees.filter(
          (employee) =>
            employee.department === selectedDepartment
        );

  const handleDepartmentChange = (event) => {
    setSelectedDepartment(event.target.value);
  };

  return (
    <div id="app-container">
      <h1 id="directory-title">Employee Directory</h1>

      <select
        id="department-filter"
        value={selectedDepartment}
        onChange={handleDepartmentChange}
      >
        <option value="All">All</option>
        <option value="Engineering">Engineering</option>
        <option value="Product">Product</option>
        <option value="Design">Design</option>
      </select>

      <div id="employee-list">
        {filteredEmployees.map((employee) => (
          <EmployeeCard
            key={employee.id}
            employeeName={employee.employeeName}
            role={employee.role}
            department={employee.department}
          />
        ))}
      </div>
    </div>
  );
}

export default App;
