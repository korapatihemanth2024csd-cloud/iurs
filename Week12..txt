2.COD weather point 
src/App.js

import React, { useState } from 'react';
import './App.css';
import CityCard from './components/CityCard';

function App() {
  const [isFahrenheit, setIsFahrenheit] = useState(false);

  const cities = [
    { city: 'Chennai', temperature: 30 },
    { city: 'Delhi', temperature: 25 },
    { city: 'Shimla', temperature: 10 }
  ];

  const toggleUnit = () => {
    setIsFahrenheit(!isFahrenheit);
  };

  return (
    <div className="app">
      <h1>Weather Point</h1>
      <p className="subtitle">City Temperatures</p>

      <button onClick={toggleUnit}>
        {isFahrenheit ? 'Show in Celsius' : 'Show in Fahrenheit'}
      </button>

      <div className="city-row">
        {cities.map((item) => {
          const temperature = isFahrenheit
            ? (item.temperature * 9 / 5) + 32
            : item.temperature;

          return (
            <CityCard
              key={item.city}
              city={item.city}
              temperature={temperature}
              unit={isFahrenheit ? 'F' : 'C'}
            />
          );
        })}
      </div>
    </div>
  );
}

export default App;

Week 12 notecraft task board
CY
src/App.js

import React, { useState } from 'react';
import './App.css';
import TaskForm from './components/TaskForm';
import TaskList from './components/TaskList';

function App() {
  const [tasks, setTasks] = useState([]);

  const addTask = (task) => {
    setTasks((prevTasks) => [...prevTasks, task]);
  };

  const removeTask = (index) => {
    setTasks((prevTasks) => prevTasks.filter((_, i) => i !== index));
  };

  return (
    <div className="app">
      <header>
        <h1>NoteCraft</h1>
        <p>Task Board</p>
      </header>

      <div className="board">
        <TaskForm onAdd={addTask} />
        <TaskList tasks={tasks} onRemove={removeTask} />
      </div>
    </div>
  );
}

export default App;

src/components/TaskForm.jsx

import React, { useState } from 'react';
import './TaskForm.css';

function TaskForm({ onAdd }) {
  const [task, setTask] = useState('');

  const handleAdd = () => {
    if (task.trim() === '') {
      return;
    }

    onAdd(task.trim());
    setTask('');
  };

  return (
    <div className="task-form">
      <input
        type="text"
        placeholder="Enter task"
        value={task}
        onChange={(e) => setTask(e.target.value)}
      />
      <button onClick={handleAdd}>Add Task</button>
    </div>
  );
}

export default TaskForm;

src/components/TaskList.jsx

import React from 'react';
import './TaskList.css';

function TaskList({ tasks, onRemove }) {
  if (tasks.length === 0) {
    return (
      <div className="task-list">
        <p className="empty-message">No tasks yet</p>
      </div>
    );
  }

  return (
    <div className="task-list">
      {tasks.map((task, index) => (
        <div className="task-row" key={index}>
          <span>{task}</span>
          <button onClick={() => onRemove(index)}>Remove</button>
        </div>
      ))}
    </div>
  );
}

export default TaskList;

2.PAH urban kitchen 
src/App.js

import React, { useState } from 'react';
import './App.css';
import SearchBar from './components/SearchBar';
import RecipeList from './components/RecipeList';

function App() {
  const [query, setQuery] = useState('');

  const recipes = [
    'Paneer Butter Masala',
    'Veg Biryani',
    'Masala Dosa',
    'Tomato Soup',
    'Paneer Tikka'
  ];

  const filteredRecipes = recipes.filter((recipe) =>
    recipe.toLowerCase().includes(query.toLowerCase())
  );

  return (
    <div className="app">
      <header>
        <h1>Urban Kitchen</h1>
        <p>Recipe Finder</p>
      </header>

      <div className="panel">
        <SearchBar query={query} onSearch={setQuery} />
        <RecipeList recipes={filteredRecipes} />
      </div>
    </div>
  );
}

export default App;
